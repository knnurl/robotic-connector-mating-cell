<< Run every line in a different terminal >>

# 1. Robot driver
ros2 launch melfa_bringup rv5as_control.launch.py use_fake_hardware:=false controller_type:="D" robot_ip:=192.168.0.20

# 2. MoveIt (move_group server, Pilz + OMPL pipelines)
ros2 launch melfa_rv5as_moveit_config rv5as_moveit.launch.py

# 3. RealSense D405 driver (also publishes camera_info used by cam_pub,
#    and the static TF camera_link -> camera_color_optical_frame)
ros2 launch realsense2_camera rs_d405_pointcloud_launch.py

# 4. Hand-eye extrinsics: TCP -> camera_link.
#    NOTE: new-style arguments. These values are the old rough guess -
#    replace with a proper hand-eye calibration (cv2.calibrateHandEye or the
#    MoveIt hand-eye calibration GUI). Roll/pitch compensation quality is
#    limited directly by the accuracy of this transform.
ros2 run tf2_ros static_transform_publisher --x 0 --y 0 --z 0 --yaw -1.5708 --pitch -1.5708 --roll 1.5708 --frame-id rv5as_default_tcp --child-frame-id camera_link

# 5. ArUco pose publisher (publishes /aruco/pose, PoseStamped in the camera
#    optical frame; view /aruco/debug_image in rqt to check detection)
ros2 run roscam cam_pub

# 6. Connector-mating controller
#    Phases: WAIT_FOR_VISION -> ALIGN_COARSE -> ALIGN_FINE -> INSERT -> MATED
#    All thresholds/speeds/offsets: mating_controller/config/rv5as_params.yaml
ros2 launch mating_controller mating_node.launch.py
